\tikzset{
  sort_fun/.style={
    fun=$<$,
    function arity=2,
  }
}

\matrix [list=a] {
  \node [elements before]; &
  \node [index=2]; &
  \node [elements between]; &
  \node [index=n]; &
  \node [elements between]; &
  \node [index=1]; &
  \node [elements between]; &
  \node [index=i]; &
  \node [elements after]; \\
};

\matrix (funs) [below=2 of list a, column sep=1em, anchor=base]{
  \node (fun1) [sort_fun]; &
  \node (fun2) [sort_fun]; &
  \node (fun3) [sort_fun]; &
  \node (fun4) [sort_fun]; &
  \node {$\ldots$}; &
  \node (fun5) [sort_fun]; \\
};

\foreach \a/\b/\r [count=\i] in {
  2/n/\true,
  n/1/\false,
  1/i/\true,
  n/i/\false,
  2/i/\false
}{
  \draw [<-] [shorten >=1ex] (fun\i.in 1) -- +(0, .5) node [anchor=base, font=\Large] {$a_\a$};
  \draw [<-] [shorten >=1ex] (fun\i.in 2) -- +(0, .5) node [anchor=base, font=\Large] {$a_\b$};
  \draw [->] [shorten >=2.5ex] (fun\i.out) -- ++(0, -.5) -- +(0, -2.5ex) node [below, anchor=base, font=\Large] {\r};
}

\node (b) [below=2 of funs] {$\{a_{min}, a_{max}\}$};

\matrix [
  matrix of math nodes,
  nodes={},
  row sep=1, 
  left delimiter=\{,
  right=of b
] {
  a_{min} \leq a_i,\, i \in [1, n] \\
  a_{max} \geq a_i,\, i \in [1, n] \\
};
