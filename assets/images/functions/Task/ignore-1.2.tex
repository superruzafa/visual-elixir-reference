\tikzset{
  task fun/.style={
    fun,
    function arity=0,
  },
}

\node (fun) [task fun];
\draw [->] (fun.out) -- +(0, -1) node [below] (b2) {$b$};
\node (process 2) [process={(fun) (b2)}];

\draw [<-] ($ (fun) + (-2.5, 0) $) coordinate (async) -- +(0, 1) coordinate (start) {};
\draw [->] (async) -- ($ (\currcoord |- process 2.south) + (0, -.25) $) coordinate (ignore);
\draw [->] (ignore) -- +(0, -.5) node (b1) [below] {\texttt{\{:ok, $b$\}}};
\node (process 1) [process={(start) (b1)}];

\draw [<-link->] ($ (fun -| process 2.west) $) -- coordinate (x) (\currcoord -| process 1.east);

\draw [--x] (ignore) -| coordinate(y) (x);

\draw [->, shorten >=2mm] (b2) |- (y);

\draw [<-note, shorten <=4mm] (async) -- +(-1.25, 0) node [font=\large,left] {\texttt{t = Task.async(fun)}};
\draw [<-note, shorten <=4mm] (ignore) -- +(-1.25, 0) node [font=\large,left,align=right] {\texttt{Task.ignore(t)}};
