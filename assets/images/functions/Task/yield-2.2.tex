\tikzset{
  task fun/.style={
    fun,
    function arity=0,
  },
}

\node (fun) [task fun];
\draw [->] (fun.out) -- +(0, -3) node [below] (b2) {$b$};
\node (process 2) [process={(fun) (b2)}] {};

\draw [<-] ($ (fun) + (-2, 0) $) coordinate (async) -- +(0, 1) coordinate (start) {};
\draw [->] (async) -- +(0, -.5) coordinate(yield);
\draw [-block-] (yield) -- ($ (\currcoord |- b2) + (0, 1) $) coordinate (timeout);
\draw [->] (timeout) -- +(0, -.5) node (nil) [below] {\nil};
\node (process 1) [process={(start) (nil)}] {};

\draw [<-link->] ($ (fun -| process 2.west) $) -| (\currcoord -| process 1.east);

\draw [<-note, shorten <=4mm] (async) -- +(-1.25, 0) node [font=\large,left] {\texttt{t = Task.async(fun)}};
\draw [<-note, shorten <=4mm] (yield) -- +(-1.25, 0) node [font=\large,left] {\texttt{Task.yield(t, timeout)}};
\draw [<-note, shorten <=4mm, align=right] (timeout) -- +(-1.25, 0) node [font=\large,left] {\texttt{timeout}};
