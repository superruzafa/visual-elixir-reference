\tikzset{
  task fun/.style={
    fun,
    function arity=0,
  },
}

\node (fun) [task fun];
\draw [->] (fun.out) -- +(0, -1) node (b2) [below] {$b$};
\node (process 2) [process={(fun) (b2)}];

\draw [<-] ($ (fun) + (-2, 0) $) coordinate (async) -- +(0, 1) coordinate (start) {};
\draw [->] (async) -- +(0, -.5) coordinate (yield);
\draw [-block-] (yield) -- ($ (\currcoord |- process 2.south) + (0, -.25) $) coordinate (res);
\draw [->] (res) -- +(0, -.5) node (b1) [below] {$\{\texttt{:ok}, b\}$};
\node (process 1) [process={(start) (b1)}];

\draw [<-link->] ($ (fun -| process 2.west) $) -| (\currcoord -| process 1.east);
\draw [->, shorten >=2mm] (b2) |- (res);

\draw [<-note, shorten <=4mm] (async) -- +(-1.25, 0) node [font=\large,left] {\texttt{t = Task.async(fun)}};
\draw [<-note, shorten <=4mm] (yield) -- +(-1.25, 0) node [font=\large,left] {\texttt{Task.yield(t)}};
